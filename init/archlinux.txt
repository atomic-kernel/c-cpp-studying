# 4. 格式化
mkfs.fat -F 32 /dev/nvme0n1p5
mkfs.ext4 /dev/nvme0n1p6
mkswap /dev/nvme0n1p7

# 5. 挂载
mount /dev/nvme0n1p6 /mnt
mount --mkdir /mnt/boot /dev/nvme0n1p5
swapon /dev/nvme0n1p7

# 6. 软件源
编辑 /etc/pacman.d/mirrorlist 仅保留 清华

# 7. 安装
pacstrap -i /mnt pacman
genfstab -U /mnt > /mnt/etc/fstab
arch-chroot -S /mnt
pacman -S grub mkinitcpio systemd
pacman -S linux linux-firmware networkmanager
pacman -S vim procps-ng git git-lfs file openssh wget curl ca-certificates vim tar gzip xz bzip2 rpm2cpio cpio iputils bind make diffutils patch gdb nano xxd util-linux zip unzip iproute2 net-tools patchelf gawk sshpass ninja autoconf automake cmake perl-mailtools perl-authen-sasl e2fsprogs
mkinitcpio -P
grub-mkconfig -o /boot/grub/grub.cfg
